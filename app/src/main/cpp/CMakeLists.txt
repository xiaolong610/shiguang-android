cmake_minimum_required(VERSION 3.22.1)
project(shiguang_books C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -Wl,-z,max-page-size=16384")
set(USE_ENCRYPTION OFF)
set(USE_XMLWRITER OFF)
set(USE_LIBXML2 OFF)
set(USE_ZLIB ON)
set(BUILD_SHARED_LIBS ON)
set(PACKAGE_VERSION "0.12")
set(PACKAGE_VERSION_MAJOR "0")
set(LIBMOBI_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/libmobi")
find_package(ZLIB REQUIRED)
add_compile_definitions(HAVE_UNISTD_H HAVE_STRDUP MOBI_INLINE=inline PACKAGE_VERSION="0.12")
add_subdirectory("${LIBMOBI_SOURCE_DIR}/src" "libmobi-build")
target_include_directories(mobi PUBLIC "${LIBMOBI_SOURCE_DIR}/src")
add_library(shiguang_books SHARED book_mobi.c)
target_link_libraries(shiguang_books mobi)
